class AddProjectDialog(QDialog):

        buttons.rejected.connect(self.reject)

        cursor = conn.cursor()

        cursor.execute("""
        INSERT INTO tasks

    def load_tasks(self):

        self.task_list.clear()

        conn = sqlite3.connect("FRIDAY_schedule.db")
        cursor = conn.cursor()

        today = QDate.currentDate().toString(
            "yyyy-MM-dd"
        )

        cursor.execute("""
        SELECT id,
            title,
            start_time,
            end_time,
            completed
        FROM tasks
        WHERE date = ?
        ORDER BY start_time
        """, (today,))

        tasks = cursor.fetchall()

        conn.close()

        for task in tasks:

            task_id, title, start, end, completed = task

            if completed:
                prefix = "☑"
            else:
                prefix = "☐"

            item = QListWidgetItem(
                f"{prefix} {start}-{end} {title}"
            )

            item.setData(
                Qt.ItemDataRole.UserRole,
                task_id
            )

            self.task_list.addItem(item)
    
    def toggle_task_completed(self, item):

        task_id = item.data(
            Qt.ItemDataRole.UserRole
        )

        conn = sqlite3.connect(
            "FRIDAYAY_schedule.db"
        )

        cursor = conn.cursor()

        cursor.execute("""
            SELECT completed
            FROM tasks
            WHERE id = ?
        """, (task_id,))

        current = cursor.fetchone()[0]

        new_value = 0 if current else 1

        cursor.execute("""
            UPDATE tasks
            SET completed = ?
            WHERE id = ?
        """, (
            new_value,
            task_id
        ))

        conn.commit()
        conn.close()

        self.load_tasks()

    def edit_task(self, item):

        task_id = item.data(
            Qt.ItemDataRole.UserRole
        )

        conn = sqlite3.connect(
            "FRIDAY_schedule.db"
        )

        cursor = conn.cursor()

        cursor.execute("""
            SELECT title, date, start_time, end_time
            FROM tasks
            WHERE id = ?
        """, (task_id,))

        task_data = cursor.fetchone()
        conn.close()

        if task_data:
            title, date, start_time, end_time = task_data

            dialog = EditTaskDialog(
                title,
                QDate.fromString(date, "yyyy-MM-dd"),
                QTime.fromString(start_time, "HH:mm"),
                QTime.fromString(end_time, "HH:mm")
            )

            if dialog.exec():
                new_title = dialog.title_input.text()
                new_date = dialog.date_input.date().toString("yyyy-MM-dd")
                new_start_time = dialog.start_input.time().toString("HH:mm")
                new_end_time = dialog.end_input.time().toString("HH:mm")

                conn = sqlite3.connect(
                    "FRIDAY_schedule.db"
                )
                cursor = conn.cursor()

                cursor.execute("""
                    UPDATE tasks
                    SET title = ?, date = ?, start_time = ?, end_time = ?
                    WHERE id = ?
                """, (
                    new_title,
                    new_date,
                    new_start_time,
                    new_end_time,
                    task_id
                ))

                conn.commit()
                conn.close()

                self.load_tasks()
 
    def task_context_menu(self, position):

        item = self.task_list.itemAt(position)

        if not item:
            return

        menu = QMenu()

        delete_action = menu.addAction(
            "Delete Task"
        )
        edit_action = menu.addAction(
            "Edit Task"
        )

        action = menu.exec(
            self.task_list.mapToGlobal(position)
        )

        if action == delete_action:

            task_id = item.data(
                Qt.ItemDataRole.UserRole
            )

            conn = sqlite3.connect(
                "FRIDAY_schedule.db"
            )

            cursor = conn.cursor()

            cursor.execute("""
                DELETE FROM tasks
                WHERE id = ?
            """, (task_id,))

            conn.commit()
            conn.close()

            self.load_tasks()
        
        elif action == edit_action:
            self.edit_task(item)
